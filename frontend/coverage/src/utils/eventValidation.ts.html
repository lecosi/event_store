// (domain/event.py and domain/session.py); the backend remains the authority.

  start_at: string // datetime-local value

export type FormErrors<T> = Partial<Record<keyof T, string>>
 
/** Parses a positive integer from an input string; NaN when it is not one. */
export function parseCapacity(value: string): number {
  return /^\d+$/.test(value.trim()) ? Number(value) : Number.NaN
}

export interface SessionFormValues {
  title: string
  speaker_id: string
  start_at: string
  end_at: string
  capacity: string
}
 
export function validateSessionForm(
  values: SessionFormValues,
  event: Pick<Event, 'start_at' | 'end_at' | 'capacity'>,
): FormErrors<SessionFormValues> {
  const errors: FormErrors<SessionFormValues> = {}
  if (!values.title.trim()) errors.title = es.validators.titleRequired
  if (!values.speaker_id) errors.speaker_id = es.validators.chooseSpeaker
 
  const start = new Date(values.start_at)
  const end = new Date(values.end_at)
  if (!values.start_at || Number.isNaN(start.getTime())) errors.start_at = es.validators.sessionStartRequired
  if (!values.end_at || Number.isNaN(end.getTime())) errors.end_at = es.validators.sessionEndRequired
  if (!errors.start_at && !errors.end_at) {
    if (end <= start) errors.end_at = es.validators.endAfterStart
    else if (start < new Date(event.start_at) || end > new Date(event.end_at))
      errors.start_at = es.validators.sessionWithinEvent
  }
 
  const capacity = parseCapacity(values.capacity)
  if (!(capacity >= 1)) errors.capacity = es.validators.capacityInvalid
  else if (capacity > event.capacity)
    errors.capacity = es.validators.capacityExceedsEvent(event.capacity)
  return errors
}
